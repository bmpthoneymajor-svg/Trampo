// ====================================================== // BT DESIGN - SCRIPT.JS // ====================================================== const API_URL = 'https://trampo.up.railway.app'; // ====================================================== // VARIÁVEIS PAYPAL // ====================================================== let paypalSdk = null; let paypalSession = null; let paypalInicializado = false; let paypalInicializando = false; // ====================================================== // CALCULAR TOTAL // ====================================================== function calcularTotal() { let total = 0; const linhas = document.querySelectorAll('.servico-row'); linhas.forEach(linha => { const checkbox = linha.querySelector('.servico-checkbox'); const quantidade = linha.querySelector('.quantidade'); const preco = linha.querySelector('.preco'); if (!checkbox || !checkbox.checked) { return; } const qtd = Number(quantidade?.value || 0); const valor = Number(preco?.value || 0); if ( Number.isFinite(qtd) && Number.isFinite(valor) ) { total += qtd * valor; } }); total = Number(total.toFixed(2)); const totalElemento = document.getElementById('total'); if (totalElemento) { totalElemento.textContent = total.toFixed(2); } return total; } // ====================================================== // ATUALIZAR TOTAL // ====================================================== document.addEventListener('input', function () { calcularTotal(); }); document.addEventListener('change', function () { calcularTotal(); }); // ====================================================== // CRIAR PEDIDO PAYPAL // ====================================================== async function criarPedidoPayPal() { const total = calcularTotal(); if (!total || total <= 0) { throw new Error( 'Selecione um serviço e informe um preço válido.' ); } console.log( 'Criando pedido PayPal:', total ); const resposta = await fetch( `${API_URL}/api/paypal/create-order`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ valor: total.toFixed(2) }) } ); const dados = await resposta.json(); if (!resposta.ok) { console.error( 'Erro PayPal:', dados ); throw new Error( dados.erro || 'Erro ao criar pedido PayPal.' ); } if (!dados.id) { throw new Error( 'O PayPal não retornou o ID do pedido.' ); } console.log( 'Pedido criado:', dados.id ); return dados.id; } // ====================================================== // CAPTURAR PAGAMENTO // ====================================================== async function capturarPedidoPayPal(orderID) { console.log( 'Capturando pedido:', orderID ); const resposta = await fetch( `${API_URL}/api/paypal/capture-order`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ orderID: orderID }) } ); const dados = await resposta.json(); console.log( 'Resposta da captura:', dados ); if (!resposta.ok) { throw new Error( dados.erro || 'Erro ao capturar pagamento.' ); } if (dados.status === 'COMPLETED') { window.location.href = '/sucesso'; return; } alert( 'Pagamento processado com status: ' + dados.status ); } // ====================================================== // MOSTRAR ERRO PAYPAL // ====================================================== function mostrarErroPayPal(mensagem) { console.error( 'PayPal:', mensagem ); let erro = document.getElementById( 'paypal-error' ); if (!erro) { erro = document.createElement('div'); erro.id = 'paypal-error'; erro.style.marginTop = '15px'; erro.style.padding = '12px'; erro.style.borderRadius = '8px'; erro.style.background = '#ffe5e5'; erro.style.color = '#a00000'; const container = document.getElementById( 'paypal-button-container' ); if (container) { container.appendChild(erro); } } if (erro) { erro.textContent = mensagem; erro.style.display = 'block'; } } // ====================================================== // INICIALIZAR PAYPAL // ====================================================== async function inicializarPayPal() { if (paypalInicializado) { return; } if (paypalInicializando) { return; } paypalInicializando = true; try { // Verificar SDK if (!window.paypal) { throw new Error( 'O SDK do PayPal ainda não foi carregado.' ); } console.log( 'Inicializando PayPal...' ); // -------------------------------------------------- // CLIENT ID // -------------------------------------------------- const resposta = await fetch( `${API_URL}/api/paypal/client-id`, { cache: 'no-store' } ); const dados = await resposta.json(); if (!resposta.ok) { throw new Error( dados.erro || 'Não foi possível obter o Client ID.' ); } if (!dados.clientId) { throw new Error( 'Client ID do PayPal não encontrado.' ); } console.log( 'Client ID recebido.' ); // -------------------------------------------------- // CRIAR INSTÂNCIA // -------------------------------------------------- paypalSdk = await window.paypal.createInstance({ clientId: dados.clientId, components: [ 'paypal-payments' ], pageType: 'checkout' }); console.log( 'Instância PayPal criada.' ); // -------------------------------------------------- // VERIFICAR ELEGIBILIDADE // -------------------------------------------------- const elegibilidade = await paypalSdk.findEligibleMethods({ currencyCode: 'BRL' }); console.log( 'Elegibilidade:', elegibilidade ); if ( !elegibilidade || !elegibilidade.isEligible('paypal') ) { throw new Error( 'O PayPal não está disponível para esta conta/configuração.' ); } // -------------------------------------------------- // LOCALIZAR BOTÃO // -------------------------------------------------- const container = document.getElementById( 'paypal-button-container' ); if (!container) { throw new Error( 'Container do PayPal não encontrado.' ); } // Limpar apenas o container PayPal container.innerHTML = ''; // -------------------------------------------------- // CRIAR BOTÃO // -------------------------------------------------- const botao = document.createElement( 'paypal-button' ); botao.id = 'paypal-btn'; botao.type = 'pay'; botao.hidden = false; container.appendChild( botao ); // -------------------------------------------------- // CRIAR SESSÃO // -------------------------------------------------- paypalSession = paypalSdk.createPayPalOneTimePaymentSession({ onApprove: async ({ orderId }) => { console.log( 'Pagamento aprovado:', orderId ); return await capturarPedidoPayPal( orderId ); }, onCancel: () => { console.log( 'Pagamento cancelado.' ); }, onError: (erro) => { console.error( 'Erro PayPal:', erro ); mostrarErroPayPal( 'Ocorreu um erro durante o pagamento.' ); } }); // -------------------------------------------------- // CLIQUE // -------------------------------------------------- botao.addEventListener( 'click', async function () { try { const total = calcularTotal(); if ( !total || total <= 0 ) { throw new Error( 'O valor do orçamento é inválido.' ); } console.log( 'Iniciando pagamento:', total ); // IMPORTANTE: // não usar await aqui. // O PayPal recomenda passar // a Promise diretamente para start(). const pedidoPromise = criarPedidoPayPal(); await paypalSession.start( { presentationMode: 'auto' }, pedidoPromise ); } catch (erro) { console.error( 'Erro ao iniciar pagamento:', erro ); mostrarErroPayPal( erro.message || 'Não foi possível iniciar o pagamento.' ); } } ); paypalInicializado = true; console.log( '================================' ); console.log( 'PAYPAL PRONTO' ); console.log( '================================' ); } catch (erro) { console.error( 'Erro ao inicializar PayPal:', erro ); mostrarErroPayPal( erro.message ); } finally { paypalInicializando = false; } } // ====================================================== // PAGAR AGORA // ====================================================== async function pagarAgora() { const total = calcularTotal(); if (!total || total <= 0) { alert( 'Selecione pelo menos um serviço e informe o preço.' ); return; } // Mostrar métodos de pagamento const paymentMethods = document.getElementById( 'payment-methods' ); if (paymentMethods) { paymentMethods.classList.remove( 'hidden' ); paymentMethods.style.display = 'block'; } // -------------------------------------------------- // PROCURAR CONTAINER PAYPAL // -------------------------------------------------- let container = document.getElementById( 'paypal-button-container' ); // -------------------------------------------------- // SE NÃO EXISTIR, CRIAR // -------------------------------------------------- if (!container) { const paymentMethodsElement = document.getElementById( 'payment-methods' ); if (!paymentMethodsElement) { alert( 'Área de pagamento não encontrada.' ); return; } const caixa = document.createElement('div'); caixa.className = 'payment-method'; caixa.innerHTML = `

Pagamento com PayPal

Total: R$ ${total.toFixed(2)}

`; paymentMethodsElement.appendChild( caixa ); container = document.getElementById( 'paypal-button-container' ); } // -------------------------------------------------- // INICIALIZAR // -------------------------------------------------- await inicializarPayPal(); } // ====================================================== // INICIALIZAÇÃO // ====================================================== document.addEventListener( 'DOMContentLoaded', function () { console.log( 'BT Design carregado.' ); calcularTotal(); } );
